  <script>
    // In-memory reviews store (resets on page refresh — for persistence, connect a backend)
    const reviews = [];

    function getSelectedRating() {
      const checked = document.querySelector('input[name="rating"]:checked');
      return checked ? parseInt(checked.value) : 0;
    }

    function starsHtml(n) {
      return '★'.repeat(n) + '☆'.repeat(5 - n);
    }

    function initials(name) {
      return name.trim().split(' ').map(w => w[0]).join('').toUpperCase().slice(0, 2) || '?';
    }

    function updateAvg() {
      if (reviews.length === 0) {
        document.getElementById('avgScore').textContent = '—';
        document.getElementById('avgStars').textContent = '☆☆☆☆☆';
        document.getElementById('avgCount').textContent = 'No reviews yet';
        return;
      }
      const avg = reviews.reduce((s, r) => s + r.rating, 0) / reviews.length;
      document.getElementById('avgScore').textContent = avg.toFixed(1);
      document.getElementById('avgStars').textContent = starsHtml(Math.round(avg));
      const n = reviews.length;
      document.getElementById('avgCount').textContent = n + ' review' + (n > 1 ? 's' : '');
    }

    function renderReviews() {
      const list = document.getElementById('reviewsList');
      const empty = document.getElementById('emptyMsg');
      if (reviews.length === 0) {
        empty.style.display = 'block';
        return;
      }
      empty.style.display = 'none';
      // rebuild
      list.innerHTML = '';
      [...reviews].reverse().forEach(r => {
        const card = document.createElement('div');
        card.className = 'review-card';
        card.innerHTML = `
          <div class="reviewer-avatar">${initials(r.name)}</div>
          <div class="reviewer-name">${r.name}${r.city ? ' · ' + r.city : ''}</div>
          <div class="review-card-stars">${starsHtml(r.rating)}</div>
          <div class="review-card-text">"${r.text}"</div>
        `;
        list.appendChild(card);
      });
    }

    function submitReview() {
      const name = document.getElementById('rName').value.trim();
      const city = document.getElementById('rCity').value.trim();
      const text = document.getElementById('rText').value.trim();
      const rating = getSelectedRating();

      if (!name) { alert('Please enter your name.'); return; }
      if (!rating) { alert('Please select a star rating.'); return; }
      if (text.length < 10) { alert('Please write at least a sentence about your experience.'); return; }

      reviews.push({ name, city, text, rating });
      updateAvg();
      renderReviews();

      // reset form
      document.getElementById('rName').value = '';
      document.getElementById('rCity').value = '';
      document.getElementById('rText').value = '';
      document.querySelectorAll('input[name="rating"]').forEach(i => i.checked = false);

      const success = document.getElementById('formSuccess');
      success.style.display = 'block';
      setTimeout(() => success.style.display = 'none', 5000);
    }

    updateAvg();
  </script>
